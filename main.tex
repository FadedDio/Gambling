% Basic definitions
\begin{frame}[t]{Some common upcoming terms}
\kick{hook}
\small
\begin{itemize}
\item \textbf{States:} The distinct conditions or situations the system can be in (e.g., money at an instant).
\item \textbf{Transitions:} The movement of the system from one state to another, or from a state back to itself (self-loop).
\item \textbf{Transition Probability:} The mathematical likelihood (between 0 and 1) that the system will move from a given current state to a specific future state.
\item \textbf{Markov Chain:} A mathematical system that models a sequence of events where the probability of the next step depends solely on the \textbf{current state}, not on how the process arrived there. This defining feature is known as the \textbf{``Markov property''} or \textbf{``memorylessness''}.
\item \textbf{Stationary Distribution:} A probability distribution over the states of a system that remains unchanged as the system evolves over time.
(\textbf{Note:} If a system begins in this distribution, its state probabilities will remain constant indefinitely.)
\item \textbf{Transition Matrix:} A square matrix where rows and columns represent states. Each element $A_{ij}$ depicts the probability to go from state $i$ to state $j$.
\end{itemize}
\end{frame}

% ------------------------------------------------------ S1 PROBLEM
\begin{frame}[t]{The setup}
\kick{Scenario 1 \ \textbullet\ \ the story}
\small She walks in with \textbf{\$k}. Every round she stakes \$1 on a biased coin: heads (prob.\ $p$) and she's up a dollar; tails (prob.\ $q$) and she's down one. She stands up the instant she hits her \textbf{target \$N} --- or busts out at \textbf{\$0}.
\vspace{0.8em}
\begin{cblock}{cTeal}{Three things we want to know}\small
  \begin{enumerate}\setlength\itemsep{3pt}
    \item What's her chance of reaching \$N before going broke?
    \item If she ditches the target and just plays until ruin --- can she get \emph{infinitely} rich?
    \item How long will she sit at the table before it's over?
  \end{enumerate}
\end{cblock}
\vspace{0.4em}
\centering\footnotesize\textit{\textcolor{cSlate}{She starts somewhere between the two walls --- and the walls decide her fate.}}
\end{frame}

% --------------------------------------------------- S1 SOLUTION I
\begin{frame}[t]{Cracking it --- her odds of winning}
\kick{Scenario 1 \ \textbullet\ \ tactic: find a closed form}
\begin{columns}[t,onlytextwidth]
  \begin{column}{0.57\textwidth}\footnotesize
    \begin{enumerate}[<+->]\setlength\itemsep{4pt}
      \item \textbf{Condition on the first bet.} From \$$k$ she lands on \$$k{+}1$ (prob.\ $p$) or \$$k{-}1$ (prob.\ $q$):
        \[ T(k)=p\,T(k{+}1)+q\,T(k{-}1). \]
      \item \textbf{Solve it.} Characteristic eq: $px^{2}-x+q=0$, roots $1$ and $q/p$. General solution:
        \[ T(k) = A(1)^k + B(q/p)^k \]
      \item \textbf{Fit the ends:} Use $T(0)=0$ and $T(N)=1$ to solve for $A$ and $B$:
        \[ p\neq q:\ T(k)=\frac{1-(q/p)^{k}}{1-(q/p)^{N}};\quad p=q:\ T(k)=\frac{k}{N}. \]
    \end{enumerate}
  \end{column}
  \begin{column}{0.41\textwidth}
    \centering
    \begin{tikzpicture}
    \begin{axis}[width=5.3cm,height=3.0cm,xlabel={start $k$},ylabel={$P(\text{reach }N)$},

  \end{column}
\end{columns}
\end{frame}

\begin{frame}[t]{Cracking it --- forever, and for how long?}
\kick{Scenario 1 \ \textbullet\ \ two more answers}
\begin{columns}[t,onlytextwidth]
  \begin{column}{0.48\textwidth}
    \begin{dblock}{cPink}{Can she get infinitely rich?}\scriptsize\color{white}
      Let $N\to\infty$.\\[2pt]
      Fair or against her $(p\le\tfrac12)$: ruin is \textbf{certain}, $T\to 0$.\\[2pt]
      Only with a true edge $(p>\tfrac12)$: $T\to 1-(q/p)^{k}>0$.\\[4pt]
      {\color{cTealL}\itshape This is ``risk of ruin'' --- why the house, on its sliver of edge, takes everyone eventually.}
    \end{dblock}
  \end{column}
  \begin{column}{0.48\textwidth}
    \begin{dblock}{cTeal}{How long at the table?}\scriptsize\color{white}
      Same trick, $+1$ for the round just played:
      \[ E_i=p\,E_{i+1}+q\,E_{i-1}+1,\ \ E_0=E_N=0. \]
      Solving this gives two distinct cases:
      \vspace{2pt}
      \textbf{If $\mathbf{p \neq q}$:}
      \[ E_i = \frac{i}{q-p} - \frac{N}{q-p} \frac{1-(q/p)^i}{1-(q/p)^N} \]
      \textbf{If $\mathbf{p = q}$ (Fair Game):}
      \[ E_i = i\,(N-i) \]
      \vspace{1pt}
      {\color{cTealL}\itshape The fair game lasts longest right in the middle.}
    \end{dblock}
  \end{column}
\end{columns}
\vspace{0.4em}
{\centering\scriptsize\textcolor{cSlate}{\textit{Where this bites in real life: position sizing, insurance reserves, bankroll management.}}\par}
\end{frame}

\dividerframe{2}{cPurple}{Scenario 2}{The Bold Gambler}{Tired of nickel-and-dime bets, she starts going all in.}

% ------------------------------------------------------ S2 PROBLEM
\begin{frame}[t]{The setup}
\kick{Scenario 2 \ \textbullet\ \ the story}
\small New attitude, new rule. With \textbf{less than half} the target $(k<\tfrac N2)$ she shoves \textbf{all} her chips --- double up (prob.\ $p$) or bust (prob.\ $q$). With \textbf{at least half} $(k\ge\tfrac N2)$ she bets exactly enough to finish, $N-k$: hit the target (prob.\ $p$), or slip back to $2k-N$ (prob.\ $q$) and keep going.
\vspace{0.6em}

\begin{columns}[t,onlytextwidth]
  \begin{column}{0.48\textwidth}
    \begin{cblock}{cPurple}{Probability of winning: $P(x)$}\scriptsize
      \[ P(x)=\begin{cases}0&x=0\\ pP(2x)&0<x<\frac{N}{2}\\ qP(2x-N)+p&\frac{N}{2}\le x<N\\ 1&x=N\end{cases} \]
    \end{cblock}
  \end{column}
  \begin{column}{0.48\textwidth}
    \begin{cblock}{cTeal}{Expected duration: $\mathbb{E}(x)$}\scriptsize
      \[ \mathbb{E}(x)=\begin{cases}0&x=0\\ 1+p\mathbb{E}(2x)&0<x<\frac{N}{2}\\ 1+q\mathbb{E}(2x-N)&\frac{N}{2}\le x<N\\ 0&x=N\end{cases} \]
    \end{cblock}
  \end{column}
\end{columns}
\vspace{0.4em}
\centering\footnotesize\textit{\textcolor{cSlate}{Both the odds of winning and expected time follow similar piecewise recursions.}}
\end{frame}

% ---------------------------------------------------- S2 SOLUTION 1
\begin{frame}[t]{Cracking it --- Probability via Binary}
\kick{Scenario 2 \ \textbullet\ \ tactic: walk the bits for $P(x)$}
\footnotesize
\begin{enumerate}[<+->]\setlength\itemsep{4pt}
  \item \textbf{Spot the binary.} Represent the ratio $\frac{k}{N}$ in binary form: $0.n_1 n_2 n_3 \dots$
  \item \textbf{Shift operations.} Doubling $2x$ is a \emph{left shift} of the digits. Losing from the top ($2x-N$) shifts \emph{and drops} the leading $1$. So the game just reads the binary string bit by bit.
  \item \textbf{Walk the bits.} To calculate $P(x)$, traverse the binary representation. For the $i^{th}$ bit:
    \begin{itemize}\setlength\itemsep{2pt}
      \item If \textbf{bit is 0}: Multiply the running probability by $p$ (equivalent to $pP(2x)$).
      \item If \textbf{bit is 1}: Multiply by $q$ and add $p$ (equivalent to $qP(2x-N)+p$).
    \end{itemize}
\end{enumerate}
\vspace{0.8em}
\begin{cblock}{cPurple}{The takeaway}\scriptsize Instead of a complicated formula, the recursion simplifies into a straightforward traversal of the binary string representing the wealth fraction.\end{cblock}
\end{frame}

% ---------------------------------------------------- S2 SOLUTION 2
\begin{frame}[t]{Cracking it --- Expected Duration via Binary}
\kick{Scenario 2 \ \textbullet\ \ tactic: walk the bits for $\mathbb{E}(x)$}
\footnotesize
\begin{enumerate}[<+->]\setlength\itemsep{4pt}
  \item \textbf{Apply the same logic.} The expected duration $\mathbb{E}(x)$ shares the exact same state transitions ($2x$ and $2x-N$). We can parse the exact same binary string to solve it.
  \item \textbf{Walk the bits.} Traverse the binary representation of $\frac{k}{N}$. For the $i^{th}$ bit:
    \begin{itemize}\setlength\itemsep{2pt}
      \item If \textbf{bit is 0}: Multiply the expectation by $p$ and add $1$ (equivalent to $1+p\mathbb{E}(2x)$).
      \item If \textbf{bit is 1}: Multiply the expectation by $q$ and add $1$ (equivalent to $1+q\mathbb{E}(2x-N)$).
    \end{itemize}
\end{enumerate}
\vspace{0.8em}
\begin{cblock}{cTeal}{The takeaway}\scriptsize The entire strategy condenses into an elegant $O(\log N)$ bit-manipulation loop. A hard problem dissolves when viewed through the right data representation.\end{cblock}
\end{frame}

\begin{frame}[t]{Cracking it --- the long-run picture}
\kick{Scenario 3 \ \textbullet\ \ tactic: balance the flow}
\begin{columns}[t,onlytextwidth]

    \vspace{-2cm} % <-- Shifted this column up to prevent cutoff
    \begin{enumerate}[<+->]\setlength\itemsep{2pt}
      \item Long-run proportions $\pi$ are unchanged by one more step: $\pi=\pi P$, where:

% -------------------------------------------------- S3 SOLUTION II
\begin{frame}[t]{Cracking it --- time to hit a target price}
\kick{Scenario 3 \ \textbullet\ \ tactic: it's just $A\mathbf{E}=\mathbf{1}$}
\footnotesize
\begin{enumerate}[<+->]\setlength\itemsep{3pt}
  \item Let $\mathbb{E}_x$ be the expected days from $x$ to $b$. Condition on the first move:
    \[ \mathbb{E}_x=p_x \mathbb{E}_{x+1}+r_x \mathbb{E}_x+q_x \mathbb{E}_{x-1}+1,\qquad \mathbb{E}_b=0. \]
  \item One equation per state. Stack them into a linear system $A\mathbf{E} = \mathbf{1}$:
    \vspace{-0.2em}
    {\scriptsize
    \[
    \begin{bmatrix}
    1-r_0 & -p_0 & 0 & \dots & 0 \\
    -q_1 & 1-r_1 & -p_1 & \dots & 0 \\
    0 & -q_2 & 1-r_2 & \dots & 0 \\
    \vdots & \vdots & \vdots & \ddots & \vdots \\
    0 & 0 & \dots & 1 & \dots \\
    \vdots & \vdots & \vdots & \vdots & \ddots \\
    0 & 0 & \dots & -q_N & 1-r_N
    \end{bmatrix}
    \begin{bmatrix} \mathbb{E}_0 \\ \mathbb{E}_1 \\ \mathbb{E}_2 \\ \vdots \\ \mathbb{E}_b \\ \vdots \\ \mathbb{E}_N \end{bmatrix}
    =
    \begin{bmatrix} 1 \\ 1 \\ 1 \\ \vdots \\ 0 \\ \vdots \\ 1 \end{bmatrix}
    \]}
    \vspace{-0.2em}
  \item Notice row $b$ is pinned entirely: $1 \cdot \mathbb{E}_b = 0$. Since the rates vary, there's no clean formula --- hand the matrix to a solver.
\end{enumerate}
\vspace{0.2em}
{\centering\scriptsize\textcolor{cSlate}{\itshape Real-world: that stationary $\pi$ is \textbf{PageRank}; a first-hitting time is expected \textbf{time-to-failure}, \textbf{time-to-churn}, or how long something takes to spread.}\par}
\end{frame}

\begin{frame}[t,shrink=5]{PageRank --- the random surfer}
\kick{Where this shows up \ \textbullet\ \ deep dive}
\small Swap the number line for a \textbf{graph of web pages}. A surfer stands on a page and, each step, clicks a \emph{random} outgoing link. Run this forever: the fraction of time spent on each page \emph{is} its rank. Same stationary-distribution math as the drifting stock --- just on a messier state space.

\begin{columns}[t,onlytextwidth]
  
  % --- LEFT HALF: TEXT ---
  \begin{column}{0.48\textwidth}\footnotesize
    \vspace{-2.5 cm} % <-- Pulls the text block up. Adjust this value to fine-tune.
    \begin{enumerate}\setlength\itemsep{0pt}
      \item \textbf{Same balance condition.} The rank vector is fixed under one more click: $\boldsymbol\pi = \boldsymbol\pi M$, where $M_{ij}$ is the chance of hopping $i\!\to\! j$.
      \item \textbf{Vote by linking.} A page is important if important pages link \emph{to} it --- importance flows along edges, split evenly across a page's out-links.
      \item \textbf{The teleport fix.} A surfer can get stuck on dead-end or cyclic pages. With prob.\ $1{-}d\ (\approx0.15)$ she \textbf{teleports} to a random page. That damping makes the chain \emph{irreducible \& aperiodic} --- so a \emph{unique} $\boldsymbol\pi$ exists and is reached from any start.
      \item \textbf{Solve by iterating.} Start uniform, apply $M$ repeatedly --- power iteration converges fast on a web-scale, billion-node graph.
    \end{enumerate}
  \end{column}

    \begin{cblock}{cTeal}{Why it mattered}\scriptsize Treating links as votes --- and ranking by the random walk's stationary distribution --- is the idea that launched Google. The same eigenvector trick now drives citation ranking, recommendation graphs, and fraud-ring detection.\end{cblock}
